cmake_minimum_required(VERSION 3.20)
project(takt_purity_public_controls LANGUAGES C)
add_executable(takt-purity-control control.c)
if(WIN32)
  target_link_libraries(takt-purity-control PRIVATE ws2_32)
else()
  add_executable(takt-purity-launcher launcher.c)
endif()
if(MSVC)
  target_compile_options(takt-purity-control PRIVATE /W4 /WX)
else()
  target_compile_options(takt-purity-control PRIVATE -Wall -Wextra -Werror)
  target_compile_options(takt-purity-launcher PRIVATE -Wall -Wextra -Werror)
endif()
